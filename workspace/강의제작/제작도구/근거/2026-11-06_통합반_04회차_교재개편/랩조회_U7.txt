## slide50.cypher
// 질문 50
MATCH (f:FailureMode)-[:REMEDIED_BY]->(s:Skill {id: 'SOP-COOL-14'})
      -[:HAS_STEP]->(t:Step)
OPTIONAL MATCH (k:Knowledge)-[c]->(t)
  WHERE type(c) <> 'STATED_IN'
RETURN s.id, t.order, s.approval_role, k.doc_id, k.rev, k.section
ORDER BY t.order;

MATCH (k:Knowledge)-[c]->(t:Step)<-[:HAS_STEP]-(s:Skill {id:'SOP-COOL-14'}) WHERE t.order=1 MATCH (k)-[:STATED_IN]->(x:Section)<-[:HAS_SECTION]-(r:Revision)<-[:HAS_REVISION]-(d:Document) RETURN k.id, k.candidate, type(c), x.id, x.ref, x.title, r.id, r.status, d.id, d.title, d.name;
// 질문 d
